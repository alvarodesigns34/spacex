import { H2R, GT3, render, spectrum, rms, db } from './run.mjs';
const SR = 48000;
function analyse(label, Cls, msg, cycleOrder, fireOrder) {
  const x = render(Cls, msg, 2.73);
  const { mag, df } = spectrum(x);
  const f0 = msg.rpm / 60;
  let tot = 0; for (const m of mag) tot += m;
  // energy within ±3 bins of each half-order harmonic (cycle = 0.5 order)
  const h = f0 * cycleOrder; let harm = 0; const orders = {};
  for (let k = 1; k * h < SR / 2 - 50; k++) {
    const c = Math.round(k * h / df); let e = 0; for (let b = c - 3; b <= c + 3; b++) e += mag[b] || 0; harm += e;
    const ord = k * cycleOrder; if ([0.5,1,1.5,2,3,4,4.5,6,8,9,12].some(o => Math.abs(o - ord) < 1e-6)) orders[ord] = e;
  }
  // band energies
  const band = (a, b) => { let e = 0; for (let i = Math.floor(a / df); i < Math.min(mag.length, b / df); i++) e += mag[i]; return e; };
  let cen = 0; for (let i = 0; i < mag.length; i++) cen += i * df * mag[i]; cen /= tot;
  const ref = orders[fireOrder];
  const ordStr = Object.entries(orders).sort((a,b)=>a[0]-b[0]).map(([o, e]) => `${o}:${db(e / ref)}`).join(' ');
  console.log(`${label.padEnd(26)} rms ${rms(x).toFixed(3)} peak ${x.reduce((a,b)=>Math.max(a,Math.abs(b)),0).toFixed(2)} centroid ${cen.toFixed(0)}Hz  inharmonic ${db(1 - harm / tot)}dB  >5k ${db(band(5000, 24000) / tot)}dB >10k ${db(band(10000,24000)/tot)}dB\n     orders rel firing: ${ordStr}`);
}
for (const [rpm, load, thr] of [[1300, 0, 0], [4000, 1, 1], [8000, 1, 1], [12000, 1, 1], [14000, 1, 1], [12000, 0, 0], [8000, 0.3, 0.3]])
  analyse(`H2R ${rpm} L${load}`, H2R, { rpm, load, thr, cut: 0, on: 1 }, 0.5, 2);
for (const [rpm, load, thr] of [[900, 0, 0], [3000, 1, 1], [6000, 1, 1], [8500, 1, 1], [8500, 0, 0], [5000, 0.3, 0.3]])
  analyse(`GT3 ${rpm} L${load}`, GT3, { rpm, load, thr, cut: 0, on: 1, gearW: 0, cabin: 0 }, 0.5, 3);
