import fs from 'fs';
import { spectrum, db, rms } from './run.mjs';
let SR = 48000;
function loadSrc(file, edits, sr) {
  const t = fs.readFileSync(file, 'utf8');
  let src = t.slice(t.indexOf('const WORKLET = `') + 17, t.indexOf('`;', t.indexOf('const WORKLET')));
  for (const [a, b] of edits) { if (!src.includes(a)) throw new Error('no match: ' + a); src = src.split(a).join(b); }
  let cls; class AWP { constructor() { this.port = {}; } }
  new Function('AudioWorkletProcessor', 'registerProcessor', 'sampleRate', src)(AWP, (n, c) => { cls = c; }, sr);
  return cls;
}
function render(Cls, msg, sec, sr) {
  const p = new Cls(); p.port.onmessage({ data: msg });
  const N = Math.floor(sr * sec), W = sr; const out = new Float32Array(N); const blk = [new Float32Array(128)];
  for (let i = 0; i < W + N; i += 128) { p.process([], [blk]); if (i >= W) out.set(blk[0].subarray(0, Math.min(128, N - (i - W))), i - W); }
  return out;
}
function inharm(x, f0, cyc, sr) {
  const { mag } = spectrum(x); const df = sr / (2 * mag.length);
  let tot = 0, harm = 0; for (let i = Math.ceil(30 / df); i < mag.length; i++) tot += mag[i];
  for (let k = 1; k * f0 * cyc < sr / 2 - 50; k++) { const c = Math.round(k * f0 * cyc / df); for (let b = c - 3; b <= c + 3; b++) harm += mag[b] || 0; }
  return db(1 - harm / tot);
}
